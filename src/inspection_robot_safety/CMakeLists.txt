cmake_minimum_required(VERSION 3.8)
project(inspection_robot_safety)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(inspection_robot_interfaces REQUIRED)
add_executable(safety_node src/safety_node.cpp)
ament_target_dependencies(safety_node rclcpp geometry_msgs sensor_msgs std_msgs
                          inspection_robot_interfaces)
install(TARGETS safety_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
